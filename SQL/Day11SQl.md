use mongo;
DELIMITER $$
CREATE FUNCTION even_odd1(n INT) RETURNS VARCHAR(20) DETERMINISTIC 
BEGIN
        IF n % 2 = 0 THEN 
        	RETURN 'Even no'; 
        ELSE 
        	RETURN 'odd';
        END IF;
END $$
DELIMITER ;
SELECT even_odd1(5);
